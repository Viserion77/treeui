Every public component in `@treeui/vue`, grouped the same way the sidebar groups them.
This index is generated from `docs/ai/treeui.catalog.json` (`pnpm ai:catalog`), so it
cannot drift from the shipped surface — component counts, categories, and links all come
from the manifests in `docs/ai/COMPONENTS/`.

The package stays intentionally composable: teams assemble forms, overlays, data screens,
and navigation from these primitives rather than from one large "kitchen sink" abstraction.
For naming and alias rules see <a target="_top" href="./?path=/docs/foundation-conventions--docs">Conventions</a>,
for the patterns TreeUI deliberately keeps as recipes see <a target="_top" href="./?path=/docs/foundation-patterns--docs">Patterns</a>,
and for the UX guarantees each component makes see <a target="_top" href="./?path=/docs/foundation-practices--docs">Practices</a>.

{(() => {
  const toId = (title) => title.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, '');
  const groups = new Map();
  for (const component of [...catalog.components].sort((a, b) => a.name.localeCompare(b.name))) {
    const category = component.category ?? 'Other';
    if (!groups.has(category)) groups.set(category, []);
    groups.get(category).push(component);
  }
  return [...groups.entries()]
    .sort((a, b) => b[1].length - a[1].length || a[0].localeCompare(b[0]))
    .map(([category, components]) => (
      <Fragment key={category}>
        <h2>{category} <small>({components.length})</small></h2>
        <p>
          {components.map((component, index) => (
            <Fragment key={component.name}>
              {index > 0 ? ' · ' : ''}
              {component.storyTitle ? (
                <a target="_top" href={`./?path=/docs/${toId(component.storyTitle)}--docs`}>
                  <code>{component.name}</code>
                </a>
              ) : (
                <code>{component.name}</code>
              )}
              {component.aliases?.length ? <> / {component.aliases.map((alias) => <code key={alias}>{alias}</code>)}</> : null}
            </Fragment>
          ))}
        </p>
      </Fragment>
    ));
})()}
